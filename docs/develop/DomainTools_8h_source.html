132  static void GetRealCoord(const PatchInfo<D>& pinfo, const std::array<int, D>& coord, std::array<double, D>& real_coord)
133  {
134  Loop::Unroll<0, D - 1>([&](int dir) {
135  if (coord[dir] == -1) {
136  real_coord[dir] = pinfo.starts[dir];
137  } else if (coord[dir] == pinfo.ns[dir]) {
138  real_coord[dir] = pinfo.starts[dir] + pinfo.spacings[dir] * pinfo.ns[dir];
139  } else {
140  real_coord[dir] = pinfo.starts[dir] + pinfo.spacings[dir] / 2.0 + pinfo.spacings[dir] * coord[dir];
141  }
142  });
143  }

168  static void GetRealCoordBound(const PatchInfo<D>& pinfo, const std::array<int, D - 1>& coord, Side<D> s, std::array<double, D>& real_coord)
169  {
170  for (size_t dir = 0; dir < s.getAxisIndex(); dir++) {
171  if (coord[dir] == -1) {
172  real_coord[dir] = pinfo.starts[dir];
173  } else if (coord[dir] == pinfo.ns[dir]) {
174  real_coord[dir] = pinfo.starts[dir] + pinfo.spacings[dir] * pinfo.ns[dir];
175  } else {
176  real_coord[dir] = pinfo.starts[dir] + pinfo.spacings[dir] / 2.0 + pinfo.spacings[dir] * coord[dir];
177  }
178  }
179  if (s.isLowerOnAxis()) {
180  real_coord[s.getAxisIndex()] = pinfo.starts[s.getAxisIndex()];
181  } else {
182  real_coord[s.getAxisIndex()] = pinfo.starts[s.getAxisIndex()] + pinfo.spacings[s.getAxisIndex()] * pinfo.ns[s.getAxisIndex()];
183  }
184  for (size_t dir = s.getAxisIndex() + 1; dir < D; dir++) {
185  if (coord[dir - 1] == -1) {
186  real_coord[dir] = pinfo.starts[dir];
187  } else if (coord[dir - 1] == pinfo.ns[dir]) {
188  real_coord[dir] = pinfo.starts[dir] + pinfo.spacings[dir] * pinfo.ns[dir];
189  } else {
190  real_coord[dir] = pinfo.starts[dir] + pinfo.spacings[dir] / 2.0 + pinfo.spacings[dir] * coord[dir - 1];
191  }
192  }
193  }
203  template<int D>
204  static void SetValues(const Domain<D>& domain, Vector<D>& vec, int component_index, std::function<double(const std::array<double, (int)D>&)> func)
205  {
206  if (component_index >= vec.getNumComponents()) {
207  throw RuntimeError("Invalid component to set");
208  }
209  std::array<double, D> real_coord;
210  for (int i = 0; i < vec.getNumLocalPatches(); i++) {
211  ComponentView<double, D> ld = vec.getComponentView(component_index, i);
212  auto pinfo = domain.getPatchInfoVector()[i];
213  Loop::Nested<D>(ld.getStart(), ld.getEnd(), [&](const std::array<int, D>& coord) {
214  GetRealCoord<D>(pinfo, coord, real_coord);
215  ld[coord] = func(real_coord);
216  });
217  }
218  }

463  template<int D>
464  static double Integrate(const Domain<D>& domain, const Vector<D>& u)
465  {
466  double sum = 0;
467 
468  for (const auto& pinfo : domain.getPatchInfoVector()) {
469  for (int c = 0; c < u.getNumComponents(); c++) {
470  ComponentView<const double, D> u_data = u.getComponentView(c, pinfo.local_index);
471 
472  double patch_sum = 0;
473  Loop::Nested<D>(u_data.getStart(), u_data.getEnd(), [&](std::array<int, D> coord) { patch_sum += u_data[coord]; });
474 
475  for (size_t i = 0; i < D; i++) {
476  patch_sum *= pinfo.spacings[i];
477  }
478  sum += patch_sum;
479  }
480  }
481  double retval;
482  MPI_Allreduce(&sum, &retval, 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
483  return retval;
484  }
